7. User-Defined Exception for Division by Zero

class DivisionByZero extends Exception {
    DivisionByZero(String message) {
        super(message);
    }
}

public class DivisionDemo {
    static int divide(int a, int b) throws DivisionByZero {
        if (b == 0)
            throw new DivisionByZero("Division by zero is not allowed.");
        return a / b;
    }

    public static void main(String[] args) {
        int a = 20;
        int b = 0;

        try {
            System.out.println("Result = " + divide(a, b));
        } catch (DivisionByZero e) {
            System.out.println("Custom Exception: " + e.getMessage());
        } finally {
            System.out.println("Finally block executed.");
        }
    }
}

8. MyThread – Constructor, super and Concurrent Execution

class MyThread extends Thread {
    MyThread(String name) {
        super(name);
        System.out.println("Created thread: " + getName());
        start();
    }

    public void run() {
        for (int i = 1; i <= 5; i++) {
            System.out.println(getName() + " : " + i);
            try {
                Thread.sleep(100);
            } catch (InterruptedException e) {
                System.out.println("Thread interrupted.");
            }
        }
    }
}

public class ThreadDemo {
    public static void main(String[] args) {
        MyThread t = new MyThread("Child Thread");

        for (int i = 1; i <= 5; i++) {
            System.out.println("Main Thread : " + i);
            try {
                Thread.sleep(100);
            } catch (InterruptedException e) {
                System.out.println("Main thread interrupted.");
            }
        }
    }
}

9. Department Enumeration using valueOf()

import java.util.Scanner;

enum Department {
    CSE, ISE, ECE, MECH
}

public class DepartmentDemo {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter department (CSE/ISE/ECE/MECH): ");
        String input = sc.nextLine().toUpperCase();

        try {
            Department dept = Department.valueOf(input);
            System.out.println("Selected Department: " + dept);
        } catch (IllegalArgumentException e) {
            System.out.println("Invalid department name.");
        }

        sc.close();
    }
}

10. Marks of Three Subjects using Wrapper Classes

import java.util.Scanner;

public class MarksDemo {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.println("Enter marks of three subjects as strings:");
        String s1 = sc.next();
        String s2 = sc.next();
        String s3 = sc.next();

        Integer m1 = Integer.valueOf(s1);
        Integer m2 = Integer.valueOf(s2);
        Integer m3 = Integer.valueOf(s3);

        int total = m1 + m2 + m3;
        double average = total / 3.0;

        System.out.println("Marks: " + m1 + ", " + m2 + ", " + m3);
        System.out.println("Total = " + total);
        System.out.println("Average = " + average);

        sc.close();
    }
}
